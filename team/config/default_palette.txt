# pgai-agent-kanban — canonical default project palette
#
# Single source of truth for the eight default project colors. Both the
# bash stack (team/scripts/lib/projects_cfg.sh, PGAI_DEFAULT_PALETTE) and
# the Python stack (team/pgai_agent_kanban/api/routers/board.py and
# projects.py, _DEFAULT_PALETTE) read this file instead of hand-
# maintaining their own copy of the array/list, so the two stacks can never
# drift apart on a project's default color.
#
# Format: one entry per data line, order is significant.
#   #RRGGBB<whitespace>name<whitespace>hue-comment
#
# Parse rule: a line is a DATA line only when its first whitespace-
# delimited token matches ^#[0-9A-Fa-f]{6}$ (a six-digit hex color).
# Every other line -- blank, or starting with anything else, including
# this header block -- is a comment and MUST be ignored by readers. This
# lets '#' double as both the hex sigil and a natural comment marker
# without ambiguity: a comment-only line never starts with a valid
# #RRGGBB token.
#
# Reading in bash:
#   while read -r hex name rest; do
#       [[ "$hex" =~ ^#[0-9A-Fa-f]{6}$ ]] || continue
#       # use $hex, $name, $rest (hue comment)
#   done < team/config/default_palette.txt
#
# Reading in Python:
#   import re
#   _HEX_RE = re.compile(r"^#[0-9A-Fa-f]{6}$")
#   for line in path.read_text().splitlines():
#       parts = line.split(None, 2)
#       if not parts or not _HEX_RE.match(parts[0]):
#           continue
#       hex_color, name, *hue_comment = parts
#
# Do not duplicate this list elsewhere -- consumers read this file; they
# do not hand-transcribe the array/list.

# ---------------------------------------------------------------------------
# Semantic dashboard color keys
# ---------------------------------------------------------------------------
#
# The eight entries above are an ORDERED list (no name field; position is
# the key) used for per-project tag colors. Below this point the SAME file
# also declares NAMED semantic keys: the color MEANING every dashboard
# renderer under team/scripts/dashboard/ and team/scripts/lib/dashboard/
# paints (halt state, deferral/approval markers, task-lifecycle status,
# and pane-framing emphasis) instead of each renderer hand-deciding "halt
# is red" or hand-transcribing its own copy of the color-name table.
#
# Parse rule for THIS section: a line is a semantic-key DATA line only
# when its first whitespace-delimited token starts with '%' -- a sigil
# that can never match the PROJECT-palette parse rule's ^#[0-9A-Fa-f]{6}$
# token, so every reader of the ordered list above safely skips this
# entire section as ordinary comment text, and the two sections coexist
# in one file without ambiguity.
#
#   %<key><whitespace><color-name><whitespace><meaning-comment>
#
# <key> is the semantic key a renderer looks up. <color-name> is one of
# the eleven names both color primitives recognize -- bash's ansi_code()
# (team/scripts/lib/script_common.sh) and Python's ansi_code()
# (pgai_agent_kanban.lib.palette) -- black, red, green, yellow, blue,
# magenta, cyan, white, bold, dim, reset. Everything after the second
# token is a free-form comment, exactly like the hue-comment column
# above.
#
# Readers:
#   bash:   team/scripts/lib/dashboard_palette.sh, dashboard_semantic_color()
#   python: pgai_agent_kanban.lib.palette.load_semantic_palette() /
#           resolve_semantic_color()
#
# A lookup of a key not declared below is a fail-loud error naming the key
# and this file -- see dashboard_semantic_color() / resolve_semantic_color()
# for the dual-channel refusal. Do not add a renderer-local fallback color
# for an undeclared key; declare the key here instead.
#
# key                    color-name   meaning
%halt_active            red          HALT sentinel active (full stop)
%halt_draining          yellow       HALT-AFTER sentinel draining
%approval_pending       yellow       Pending HUMAN-APPROVE gate marker
%defer_warning          yellow       Deferral posture unreadable/malformed
%status_open            white        Task/entry lifecycle: open (default)
%status_running         yellow       Task/entry lifecycle: running (amber)
%status_done            green        Task/entry lifecycle: done / wont-do
%status_blocked         red          Task/entry lifecycle: blocked
%status_waiting         yellow       Queue lifecycle: waiting
%status_backlog         dim          Queue lifecycle: backlog
%frame_label            cyan         Column/pane header label text
%frame_dim              dim          Secondary/faint framing text
%frame_bold             bold         Emphasized framing text
%frame_warn             yellow       Framing warning marker
