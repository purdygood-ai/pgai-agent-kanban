# overwatch_resolve: this project's override of whether OVERWATCH's
# resolution stage may run its lanes on a CM ship-policy HALT its matrix
# does not resolve, rather than leave that HALT to the matrix alone — read
# by pp_overwatch_resolve() (team/scripts/lib/project_paths.sh) and its
# Python counterpart
# resolve_overwatch_resolve() (team/pgai_agent_kanban/overwatch/
# resolution_matrix.py). Checked here first; falls back to the kanban-wide
# default in kanban.cfg when commented out.
#
# Turning this on authorises OVERWATCH to execute its documented recovery
# playbooks (Lane A: rewrite-and-recycle a bad-fix regression; Lane B: ship
# a HALT whose cause matches a declared operator INTENT.md entry) for THIS
# project without waiting for a human — see team/roles/OVERWATCH.md,
# "Autonomous Block Resolution". Everything else (Lane C) becomes one
# operator-inbox entry reading the cause and the options; nothing is
# executed. Whatever this key is set to, the blocked CM task behind a HALT
# the stage leaves in place is decided like any other block: its
# classifier row's verb while the row's budget allows, then its
# operator-inbox question, with Needs Human cleared.
#
# Left commented out so a fresh project ships dark: absent, blank, or
# unreadable resolves to false — never an error, never "true". Only an
# unrecognized non-boolean value (a typo, 'yes') refuses loud.
# Accepted: true|false
# Inherits: project.cfg > kanban.cfg > false
# Default: false
# overwatch_resolve = false
